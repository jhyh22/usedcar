<script>

function generateOneSet() {

    let numbers = [];

    for (let i = 1; i <= 45; i++) {
        numbers.push(i);
    }

    // Fisher-Yates 셔플
    for (let i = numbers.length - 1; i > 0; i--) {

        const j = Math.floor(Math.random() * (i + 1));

        [numbers[i], numbers[j]] =
        [numbers[j], numbers[i]];
    }

    // 앞의 6개 = 본 번호
    const mainNumbers =
        numbers.slice(0, 6).sort((a, b) => a - b);

    // 7번째 = 보너스 번호
    const bonusNumber = numbers[6];

    return {
        main: mainNumbers,
        bonus: bonusNumber
    };
}


function getBallColor(number) {

    if (number <= 10) {
        return "yellow";
    }

    if (number <= 20) {
        return "blue";
    }

    if (number <= 30) {
        return "red";
    }

    if (number <= 40) {
        return "gray";
    }

    return "green";
}


function generateLotto() {

    const rows =
        document.querySelectorAll(".lotto-row");

    rows.forEach((row) => {

        const numbersContainer =
            row.querySelector(".numbers");

        numbersContainer.innerHTML = "";

        const result = generateOneSet();

        // 본 번호 6개
        result.main.forEach((number, index) => {

            const ball =
                document.createElement("div");

            ball.className =
                "ball " + getBallColor(number);

            ball.textContent = number;

            ball.style.animationDelay =
                (index * 0.06) + "s";

            numbersContainer.appendChild(ball);
        });


        // BONUS 영역
        const bonusArea =
            document.createElement("div");

        bonusArea.className =
            "bonus-area";


        const bonusLabel =
            document.createElement("span");

        bonusLabel.className =
            "bonus-label";

        bonusLabel.textContent =
            "BONUS";


        const bonusBall =
            document.createElement("div");

        bonusBall.className =
            "ball bonus-ball " +
            getBallColor(result.bonus);

        bonusBall.textContent =
            result.bonus;

        bonusBall.style.animationDelay =
            "0.4s";


        bonusArea.appendChild(bonusLabel);

        bonusArea.appendChild(bonusBall);

        numbersContainer.appendChild(bonusArea);

    });
}


// 페이지를 처음 열었을 때 자동으로 번호 표시
generateLotto();

</script>
